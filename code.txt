func classify(n) {
    if (n < 0) {
        return -1;
    } else if (n == 0) {
        return 0;
    } else if (n < 10) {
        if (n == 7) {
            return 777;
        } else {
            return 1;
        }
    } else if (n < 100) {
        return 2;
    } else {
        return 3;
    }
}

print classify(-5); print " ";
print classify(0); print " ";
print classify(4); print " ";
print classify(7); print " ";
print classify(42); print " ";
print classify(500); print " | ";
