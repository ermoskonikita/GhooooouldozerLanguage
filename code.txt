// 1. Скалярная арифметика и проверка порядка операндов (-= и /=)
a = 20;
a -= 6;      // 20 - 6 = 14 (если порядок перепутан, выйдет -14)
a *= 3;      // 14 * 3 = 42
a /= 7;      // 42 / 7 = 6  (если порядок перепутан, выйдет 0)
print a;
print "\n";

// 2. Побитовые составные операторы (&=, |=, ^=)
mask = 12;   // 1100
mask &= 10;  // 1100 & 1010 = 1000 (8)
print mask;
print "\n";

mask |= 3;   // 1000 | 0011 = 1011 (11)
print mask;
print "\n";

mask ^= 15;  // 1011 ^ 1111 = 0100 (4)
print mask;
print "\n";

// 3. Правоассоциативная цепочка со смешанными операторами
x = 100;
y = 20;
z = 3;
// Справа налево: z = 6 -> y = 14 -> x = 100 / 14 = 7
x /= y -= z *= 2;
print x;
print "\n";
print y;
print "\n";
print z;
print "\n";

// 4. Массивы: некоммутативные и побитовые операторы + цепочки
array arr[4];
arr[0] = 50;
arr[1] = 10;
arr[2] = 15;
arr[3] = 5;

i = 0;
arr[i] -= 15;          // 50 - 15 = 35
arr[i + 1] /= 2;       // 10 / 2 = 5
arr[2] &= 6;           // 15 & 6 = 6
arr[3] ^= 3;           // 5 ^ 3 = 6

// Цепочка в массивах: arr[1] = 5 | 8 = 13 -> arr[0] = 35 - 13 = 22
arr[0] -= arr[1] |= 8;

print arr[0];
print "\n";
print arr[1];
print "\n";
print arr[2];
print "\n";
print arr[3];
print "\n";

// 5. Проверка однократного вычисления индекса (побочные эффекты)
array side[3];
side[0] = 100;
side[1] = 200;
side[2] = 300;

idx = 0;
side[idx += 1] += 50; // idx должен стать 1 (не 2!), side[1] = 250
print idx;
print "\n";
print side[1];
print "\n";
print side[2];
print "\n";

side[idx += 1] = 999; // idx должен стать 2, side[2] = 999
print idx;
print "\n";
print side[2];
print "\n";

// Бонус: вложенный побочный эффект в индексе (рекурсивный check_vector)
// side[0] становится 100 - 99 = 1 -> side[1] становится 250 + 11 = 261
side[side[0] -= 99] += 11;
print side[0];
print "\n";
print side[1];
print "\n";
