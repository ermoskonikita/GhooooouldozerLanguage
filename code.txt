print "--- Functions & Arrays Test ---\n";

// 1. ОБЪЯВЛЯЕМ ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ
size = 3;
array arr[size * 2];
multiplier = 2;

// 2. ОБЪЯВЛЯЕМ ФУНКЦИИ
func fill_array() {
    idx = 0;
    while (idx < 6) {
        arr[idx] = idx * 10;
        idx++;
    }
    return 1;
}

func double_array() {
    idx = 0;
    while (idx < 6) {
        arr[idx] = arr[idx] * multiplier;
        idx++;
    }
    return 1;
}

func get_magic() {
    return 99;
}

// 3. ОСНОВНАЯ ПРОГРАММА
print "Calling fill_array()...\n";
status = fill_array();

print "arr[2] before (expected 20) = ";
print arr[2];
print "\n";

print "Calling double_array()...\n";
status = double_array();

print "arr[2] after (expected 40) = ";
print arr[2];
print "\n";

print "Calling get_magic()...\n";
magic = get_magic();
print "magic (expected 99) = ";
print magic;
print "\n";

// 4. ТЕСТ ИНКРЕМЕНТОВ И МАССИВОВ
print "--- Increments ---\n";
x = arr[2]++; 
y = ++arr[3]; 

print "x (expected 40) = ";
print x;
print "\n";

print "arr[2] (expected 41) = ";
print arr[2];
print "\n";

print "y (expected 61) = ";
print y;
print "\n";

// 5. ТЕСТ ВЕТВЛЕНИЙ
print "--- If/Else ---\n";
if (arr[2] > 40) {
    print "arr[2] is > 40 (CORRECT)\n";
} else {
    print "ERROR!\n";
}

// 6. ТЕСТ ОДИНОЧНЫХ ИНКРЕМЕНТОВ
arr[0]++;
++arr[1];

print "arr[0] (expected 1) = ";
print arr[0];
print "\n";

print "arr[1] (expected 21) = ";
print arr[1];
print "\n";

print "ALL TESTS PASSED SUCCESSFULLY!\n";
