// === 1. Объявление функций ===
func getBitMask() {
    return 15; // 0b00001111
}

// === 2. Побитовые операции (&, |, ^, ~) ===
a = 12; // 0b00001100
b = 5;  // 0b00000101

// BAND (12 & 5 = 4)
res_band = a & b;
print res_band; // Ожидается: 4

// BOR (12 | 5 = 13)
res_bor = a | b;
print res_bor; // Ожидается: 13

// BXOR (12 ^ 5 = 9)
res_bxor = a ^ b;
print res_bxor; // Ожидается: 9

// BNOT (~0 = -1)
res_bnot = ~0;
print res_bnot; // Ожидается: -1

// === 3. Вызов функции внутри побитового выражения ===
// 12 & 15 = 12
mask_res = a & getBitMask();
print mask_res; // Ожидается: 12

// === 4. Проверка приоритета: a | b & c ^ d ===
// По правилам: b & c (5 & 12 = 4) -> 4 ^ 3 (7) -> 12 | 7 (15)
pri_test = 12 | 5 & 12 ^ 3;
print pri_test; // Ожидается: 15

// === 5. Сложная логика с математикой и вызовами ===
x = 10;
y = 0;
if ((x & 2) == 2 && (y != 0 || !y)) {
    print 111; // Ожидается: 111
} else {
    print 0;
}

// === 6. Цикл While с побитовой маской ===
i = 1;
while (i <= 8) {
    // Печатаем i, если включен бит 2 или бит 8 (i & 10 != 0)
    if ((i & 10) != 0) {
        print i; // Ожидается: 2, 8
    }
    i = i * 2;
}

print 999; // Финальный маркер
