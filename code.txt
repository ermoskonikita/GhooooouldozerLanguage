// 1. Арифметика, приоритет и составные присваивания
a = 10;
b = 3;
res1 = -a + b * 4 - (10 % 3); // -10 + 12 - 1 = 1

a += 5;  // 15
a -= 2;  // 13
a *= 3;  // 39
a /= 4;  // 9
a %= 6;  // 3

print "1. Arith: ";
print res1; print " "; print a; print "\n";

// 2. Побитовые операции и сдвиги (включая арифметический >>)
mask = 5;       // 0101
mask <<= 3;     // 40 (101000)
mask |= 3;      // 43 (101011)
mask ^= 10;     // 33 (100001)
mask &= 63;     // 33
mask >>= 2;     // 8

neg = -16;
neg >>= 2;      // -4
inv = ~0;       // -1

print "2. Bits: ";
print mask; print " "; print neg; print " "; print inv; print "\n";

// 3. Сравнения и ленивые && / || (защита от деления на 0)
z = 0;
sc1 = (z != 0) && ((10 / z) > 1); // 0
sc2 = (z == 0) || ((10 / z) > 1); // 1
sc3 = !(5 <= 3) && (4 >= 4);      // 1

print "3. Logic: ";
print sc1; print " "; print sc2; print " "; print sc3; print "\n";

// 4. Инкремент/декремент на переменных и массивах
c = 5;
p1 = c++; // p1 = 5, c = 6
p2 = ++c; // p2 = 7, c = 7

array nums[3];
nums[0] = 10;
nums[1] = 20;
nums[2] = 30;

p3 = nums[1]++; // p3 = 20, nums[1] = 21
p4 = --nums[2]; // p4 = 29, nums[2] = 29
nums[0] <<= 1;  // nums[0] = 20

print "4. Inc/Arr: ";
print p1; print " "; print p2; print " ";
print p3; print " "; print p4; print " ";
print nums[0]; print " "; print nums[1]; print " ";
print length(nums); print "\n";

// 5. Ветвления и циклы (for, while, continue, break, пустые секции)
sum = 0;
for (i = 0; i < 10; i++) {
    if (i % 2 == 0) {
        continue; // пропускаем 0, 2, 4, 6, 8; шаг i++ должен сработать!
    }
    else if (i == 9) {
        break;
    }
    else {
        sum += i; // 1 + 3 + 5 + 7 = 16
    }
}

w = 0;
while (w < 5) {
    w++;
}
for (; w < 8; ) {
    w++;
}

print "5. Loops: ";
print sum; print " "; print i; print " "; print w; print "\n";

// 6. Функции, рекурсия, глобальные переменные и передача массивов
global_acc = 100;

func fact(n) {
    if (n <= 1) {
        return 1;
    }
    return n * fact(n - 1);
}

func fill_and_sum(array arr, mult) {
    local_s = 0;
    for (k = 0; k < length(arr); k++) {
        arr[k] = (k + 1) * mult;
        local_s += arr[k];
    }
    global_acc += local_s;
    return local_s;
}

f5 = fact(5);
array data[4];
arr_sum = fill_and_sum(data, 10);

print "6. Funcs: ";
print f5; print " "; print arr_sum; print " ";
print data[3]; print " "; print global_acc; print "\n";
