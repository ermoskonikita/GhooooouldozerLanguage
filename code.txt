// 1. Скалярная арифметика и проверка порядка операндов (-= и /=)
a = 20;
a -= 6;      // 20 - 6 = 14 (если порядок перепутан, выйдет -14)
a *= 3;      // 14 * 3 = 42
a /= 7;      // 42 / 7 = 6  (если порядок перепутан, выйдет 0)
print a;

// 2. Побитовые составные операторы (&=, |=, ^=)
mask = 12;   // 1100
mask &= 10;  // 1100 & 1010 = 1000 (8)
print mask;

mask |= 3;   // 1000 | 0011 = 1011 (11)
print mask;

mask ^= 15;  // 1011 ^ 1111 = 0100 (4)
print mask;

// 3. Правоассоциативная цепочка со смешанными операторами
x = 100;
y = 20;
z = 3;
// Справа налево: z = 6 -> y = 14 -> x = 100 / 14 = 7
x /= y -= z *= 2;
print x;
print y;
print z;

// 4. Массивы: некоммутативные и побитовые операторы + цепочки
array arr[4];
arr[0] = 50;
arr[1] = 10;
arr[2] = 15;
arr[3] = 5;

i = 0;
arr[i] -= 15;          // 50 - 15 = 35
arr[i + 1] /= 2;       // 10 / 2 = 5
arr[2] &= 6;           // 15 & 6 = 6
arr[3] ^= 3;           // 5 ^ 3 = 6

// Цепочка в массивах: arr[1] = 5 | 8 = 13 -> arr[0] = 35 - 13 = 22
arr[0] -= arr[1] |= 8;

print arr[0];
print arr[1];
print arr[2];
print arr[3];
