// --- Главная программа ---
PUSH 5        // Кладем аргумент для функции (факториал какого числа ищем)
CALL factorial // Вызываем функцию факториала
PRINT         // Печатаем результат (должно вывести 120)
HALT          // Конец программы

// --- Функция вычисления факториала (рекурсивная) ---
factorial:
    // На стеке лежит n. Проверим, равно ли оно 0 или 1.
    // Для этого сохраним копию n и сравним с 1 (или 0)
    DUP
    PUSH 1
    EQ
    JNZ base_case   // Если n == 1, прыгаем на базовый случай

    // Шаг рекурсии: возвращаем n * factorial(n - 1)
    DUP
    PUSH 1
    SUB
    CALL factorial // Рекурсивный вызов
    MUL            // Умножаем n на результат factorial(n - 1)
    RET            // Возврат из функции

base_case:
    // Базовый случай: если n == 1, факториал равен 1. 
    // Очищаем лишний аргумент со стека и возвращаем 1.
    POP            // Убираем n со стека
    PUSH 1         // Класть результат 1
    RET            // Возврат из функции
